
QNO 4:
A university wants to track students' exam results. If a student is deleted, their results should also be deleted.
Tasks:
Create a Students table:
StudentID (PRIMARY KEY, AUTO_INCREMENT)
StudentName (NOT NULL)

mysql> create table student4(studentid int primary key auto_increment,student_name varchar(50) not null);

| student_name | varchar(50) | NO   |     | NULL    |                |
+--------------+-------------+------+-----+---------+----------------+



Create an Exams table:

ExamID (PRIMARY KEY, AUTO_INCREMENT)
SubjectName (NOT NULL, UNIQUE)




mysql> create table exam4(examid int primary key auto_increment,subject_name varchar(50) not null unique);
Query OK, 0 rows affected (0.03 sec)

mysql> desc exam4;
+--------------+-------------+------+-----+---------+----------------+
| Field        | Type        | Null | Key | Default | Extra          |
+--------------+-------------+------+-----+---------+----------------+
| examid       | int         | NO   | PRI | NULL    | auto_increment |
| subject_name | varchar(50) | NO   | UNI | NULL    |                |
+--------------+-------------+------+-----+---------+----------------+





Create a Results table:


ResultID (PRIMARY KEY, AUTO_INCREMENT)
StudentID (FOREIGN key)
ExamID (FOREIGN KEY)
Score (NOT NULL, CHECK Score BETWEEN 0 AND 100)

mysql> create table result4(resultid int primary key auto_increment,studentid int,examid int,foreign key(studentid) references student4(studentid) on delete cascade,foreign key(examid) references exam4(examid),score decimal(10,2) check (score between 0 and 100));
